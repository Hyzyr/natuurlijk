Gebruiksinstructies

1. Include het script in je HTML

<script src="/js/client-info.js"></script>

2. Gebruik data-attributes voor automatische updates

<!-- Eenvoudige tekst -->
<span data-client-info="contact.email"></span>

<!-- Link met e-mail -->
<a href="mailto:" 
   data-client-info="contact.email" 
   data-client-info-attr="href:mailto:{value}">
  <span data-client-info="contact.email"></span>
</a>

<!-- Telefoon link -->
<a href="tel:" 
   data-client-info="contact.phone" 
   data-client-info-attr="href:tel:{value}">
  <span data-client-info="contact.phoneDisplay"></span>
</a>

3. Initialiseer op pagina load

document.addEventListener('DOMContentLoaded', async () => {
  await clientInfo.init();
});

4. Of gebruik programmatisch

// Laad de data
await clientInfo.load();

// Verkrijg specifieke informatie
const contact = clientInfo.getContact();
console.log('E-mail:', contact.email);

const shipping = clientInfo.getShipping();
console.log('Gratis verzending vanaf:', shipping.freeShippingThreshold);
